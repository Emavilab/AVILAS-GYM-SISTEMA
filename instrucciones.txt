-- Crear base de datos --
CREATE DATABASE gym_db;

-- Crear tablas --
CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario VARCHAR(50),
    password VARCHAR(255)
);

CREATE TABLE clientes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100),
    telefono VARCHAR(20),
    fecha_registro DATE,
    fecha_vencimiento DATE,
    estado ENUM('activo', 'vencido') DEFAULT 'activo'
);
